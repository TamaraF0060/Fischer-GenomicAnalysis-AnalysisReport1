grep -F -f dup.fam ../raw/hw2.fam > hw2_duplicates.fam

# -F says to treat the search term as literal text, not a pattern (avoids issues with special characters)
# -f dup.fam says to use this file for the “patterns” to search for

wc -l hw2_duplicates.fam
# there are 147 duplicates
# when I originally tried to remove the duplicates from hw2.fam like this:
# grep -v -f hw2_duplicates.fam ../raw/hw2.fam > dups_removed.fam
# the math was not mathing ugh! so I had to ask chat if I was doing it right
# chat said grep removes any lines with matching patterns, INCLUDING the original occurrence and suggested using awk as a better way to remove duplicates like this:
awk '!seen[$2]++' ../raw/hw2.fam > hw2_deduplicated.fam
# seen[$2] creates a tracking table (counts the number of times a value is seen)
# seen[$2]++ increases the count every time the value is seen
# !seen[$2]++ keeps the first occurrence and removes all later duplicates
# duplicates are now removed and we can count how many times each unique FID is in the file

cut -f1 hw2_deduplicated.fam | sort | uniq -c > hw2_FID.counts
# uniq -c counts consecutive (have to sort first) duplicates
